cmake_minimum_required(VERSION 3.31.6)
# CMP0091 queda NEW antes de habilitar el compilador.
project(NppLdapFormatter VERSION 0.1.0 LANGUAGES NONE)
include(cmake/Baseline.cmake)

option(NPP_LDAP_MANIFEST_ONLY "Validar SDK/manifiesto sin compilar C++" OFF)
set(NPP_LDAP_SDK_ROOT "${CMAKE_SOURCE_DIR}/.build/notepad-plus-plus" CACHE PATH "Checkout fijado de headers oficiales")
if(NOT NPP_LDAP_MANIFEST_ONLY)
    if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Visual Studio 17 2022")
        message(FATAL_ERROR "Baseline nativo: Windows y Visual Studio 17 2022.")
    endif()
    # CMAKE_GENERATOR_PLATFORM también contiene version=<SDK>; validar el nombre resuelto.
    if(NOT CMAKE_VS_PLATFORM_NAME MATCHES "^(Win32|x64|ARM64)$")
        message(FATAL_ERROR "Seleccione Win32, x64 o ARM64 mediante un preset.")
    endif()
    if(NOT CMAKE_VS_PLATFORM_TOOLSET STREQUAL "v143")
        message(FATAL_ERROR "Se requiere toolset v143.")
    endif()
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    enable_language(CXX)
    if(NOT MSVC OR NOT CMAKE_CXX_COMPILER_VERSION MATCHES "^19\\.44\\.")
        message(FATAL_ERROR "Se requiere MSVC 19.44 (toolset 14.44).")
    endif()
    if(NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL NPP_LDAP_WINDOWS_SDK)
        message(FATAL_ERROR "Windows SDK debe ser ${NPP_LDAP_WINDOWS_SDK}.")
    endif()
    # CMake puede dejar CMAKE_VS_PLATFORM_TOOLSET_VERSION vacío al resolver el
    # toolset predeterminado; comprobar la solicitud y el directorio real de cl.exe.
    if(NOT CMAKE_GENERATOR_TOOLSET STREQUAL "v143,version=${NPP_LDAP_TOOLSET_VERSION}")
        message(FATAL_ERROR "Seleccione v143,version=${NPP_LDAP_TOOLSET_VERSION} mediante un preset.")
    endif()
    get_filename_component(msvc_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(parent IN ITEMS 1 2 3)
        get_filename_component(msvc_directory "${msvc_directory}" DIRECTORY)
    endforeach()
    get_filename_component(msvc_directory_version "${msvc_directory}" NAME)
    if(NOT msvc_directory_version STREQUAL NPP_LDAP_TOOLSET_VERSION)
        message(FATAL_ERROR "Toolset debe ser ${NPP_LDAP_TOOLSET_VERSION}; cl.exe está en '${msvc_directory_version}'.")
    endif()
endif()

find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${NPP_LDAP_SDK_ROOT}" rev-parse HEAD
    RESULT_VARIABLE sdk_result OUTPUT_VARIABLE sdk_revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT sdk_result EQUAL 0 OR NOT sdk_revision STREQUAL NPP_LDAP_SDK_REVISION)
    message(FATAL_ERROR "SDK ausente o revisión distinta: se requiere ${NPP_LDAP_SDK_REVISION}.")
endif()
# El commit no basta: rechazar headers modificados o añadidos al SDK consumido.
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${NPP_LDAP_SDK_ROOT}" status
    --porcelain --untracked-files=all -- PowerEditor/src/MISC/PluginsManager scintilla/include
    RESULT_VARIABLE sdk_dirty_result OUTPUT_VARIABLE sdk_dirty OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT sdk_dirty_result EQUAL 0 OR NOT sdk_dirty STREQUAL "")
    message(FATAL_ERROR "SDK modificado: los headers deben coincidir con el commit fijado.")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${NPP_LDAP_SDK_ROOT}" show HEAD:scintilla/version.txt
    RESULT_VARIABLE sci_result OUTPUT_VARIABLE sci_version OUTPUT_STRIP_TRAILING_WHITESPACE)
string(REPLACE "." "" expected_sci_version "${NPP_LDAP_SCINTILLA_VERSION}")
if(NOT sci_result EQUAL 0 OR NOT sci_version STREQUAL expected_sci_version)
    message(FATAL_ERROR "La versión Scintilla del SDK no coincide con Baseline.cmake.")
endif()
set(npp_headers "${NPP_LDAP_SDK_ROOT}/PowerEditor/src/MISC/PluginsManager")
set(sci_headers "${NPP_LDAP_SDK_ROOT}/scintilla/include")
foreach(header IN ITEMS "${npp_headers}/PluginInterface.h" "${npp_headers}/Notepad_plus_msgs.h" "${sci_headers}/Scintilla.h" "${sci_headers}/Sci_Position.h")
    if(NOT EXISTS "${header}")
        message(FATAL_ERROR "Header requerido ausente: ${header}")
    endif()
endforeach()
file(READ "${npp_headers}/Notepad_plus_msgs.h" npp_text)
file(READ "${sci_headers}/Scintilla.h" sci_text)
foreach(api IN ITEMS NPPM_GETCURRENTSCINTILLA NPPM_GETCURRENTBUFFERID)
    if(NOT npp_text MATCHES "#define[ \t]+${api}[ \t]")
        message(FATAL_ERROR "API Notepad++ requerida ausente: ${api}")
    endif()
endforeach()
foreach(api IN ITEMS SCI_GETTEXTRANGEFULL SCI_GETCODEPAGE SCI_BEGINUNDOACTION SCI_REPLACESEL SCI_ENDUNDOACTION SCI_ADDUNDOACTION SC_LASTSTEPINUNDOREDO SC_CP_UTF8)
    if(NOT sci_text MATCHES "#define[ \t]+${api}[ \t]")
        message(FATAL_ERROR "API Scintilla requerida ausente: ${api}")
    endif()
endforeach()

if(NOT NPP_LDAP_MANIFEST_ONLY)
    add_library(npp_ldap_build_options INTERFACE)
    target_compile_features(npp_ldap_build_options INTERFACE cxx_std_20)
    target_compile_definitions(npp_ldap_build_options INTERFACE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(npp_ldap_build_options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc)
    add_library(npp_ldap_host_sdk INTERFACE)
    target_include_directories(npp_ldap_host_sdk INTERFACE "${npp_headers}" "${sci_headers}")
    target_link_libraries(npp_ldap_host_sdk INTERFACE npp_ldap_build_options user32)
    # La sonda valida headers independientemente del scaffold DLL.
    add_library(baseline_sdk_probe STATIC cmake/sdk_probe.cpp)
    target_link_libraries(baseline_sdk_probe PRIVATE npp_ldap_host_sdk)

    add_library(ldap_core STATIC src/BuildInfo.cpp src/LdapParser.cpp src/CoreBudget.cpp)
    target_include_directories(ldap_core PUBLIC include)
    target_compile_features(ldap_core PUBLIC cxx_std_20)
    target_compile_options(ldap_core PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    add_library(ldap_gateway STATIC src/ScintillaGateway.cpp)
    target_include_directories(ldap_gateway PUBLIC include)
    target_link_libraries(ldap_gateway PUBLIC npp_ldap_host_sdk)
    add_library(NppLdapFormatter MODULE src/PluginDefinition.cpp
        src/PluginExports.cpp src/PluginExports.def)
    target_include_directories(NppLdapFormatter PRIVATE include)
    target_link_libraries(NppLdapFormatter PRIVATE ldap_core ldap_gateway)
    target_link_options(NppLdapFormatter PRIVATE /WX)
    set_target_properties(NppLdapFormatter PROPERTIES PREFIX "")

    include(CTest)
    if(BUILD_TESTING)
        add_executable(ldap_parser_tests tests/LdapParserTests.cpp)
        target_link_libraries(ldap_parser_tests PRIVATE ldap_core)
        target_compile_options(ldap_parser_tests PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
        add_test(NAME ldap_parser_tests COMMAND ldap_parser_tests)
        add_executable(plugin_abi_smoke tests/PluginAbiSmoke.cpp)
        target_link_libraries(plugin_abi_smoke PRIVATE npp_ldap_host_sdk)
        add_dependencies(plugin_abi_smoke NppLdapFormatter)
        add_test(NAME plugin_abi_smoke
            COMMAND plugin_abi_smoke "$<TARGET_FILE:NppLdapFormatter>")
    endif()
endif()
message(STATUS "Baseline: Npp ${NPP_LDAP_HOST_VERSION}, SDK ${sdk_revision}, C++20, CRT estático")
